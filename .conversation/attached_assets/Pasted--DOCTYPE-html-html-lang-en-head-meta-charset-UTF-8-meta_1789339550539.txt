<script>

  /* =========================
     DATA
  ========================= */

  const STORAGE_KEY = "macrotrack_data";

  const defaultData = {
    goals: {
      calories: 2500,
      protein: 150,
      carbs: 300,
      fat: 80
    },

    days: {},

    savedFoods: []
  };


  let data = loadData();

  const todayKey = getDateKey(new Date());

  let currentDate = todayKey;


  /* =========================
     STORAGE
  ========================= */

  function loadData() {

    const saved = localStorage.getItem(STORAGE_KEY);

    if (!saved) {
      return JSON.parse(JSON.stringify(defaultData));
    }

    try {

      const parsed = JSON.parse(saved);

      return {
        goals: {
          ...defaultData.goals,
          ...(parsed.goals || {})
        },

        days: parsed.days || {},

        savedFoods: parsed.savedFoods || []
      };

    } catch {

      return JSON.parse(JSON.stringify(defaultData));

    }
  }


  function saveData() {
    localStorage.setItem(
      STORAGE_KEY,
      JSON.stringify(data)
    );
  }


  /* =========================
     DATE
  ========================= */

  function getDateKey(date) {

    const year = date.getFullYear();

    const month = String(
      date.getMonth() + 1
    ).padStart(2, "0");

    const day = String(
      date.getDate()
    ).padStart(2, "0");

    return `${year}-${month}-${day}`;
  }


  function formatDate(key) {

    const parts = key.split("-");

    const date = new Date(
      Number(parts[0]),
      Number(parts[1]) - 1,
      Number(parts[2])
    );

    return date.toLocaleDateString(
      undefined,
      {
        weekday: "long",
        month: "long",
        day: "numeric",
        year: "numeric"
      }
    );
  }


  /* =========================
     DAY
  ========================= */

  function getCurrentDay() {

    if (!data.days[currentDate]) {

      data.days[currentDate] = {
        foods: []
      };

    }

    return data.days[currentDate];
  }


  /* =========================
     TOTALS
  ========================= */

  function getTotals() {

    const foods =
      getCurrentDay().foods;

    return foods.reduce(
      (total, food) => {

        total.calories += Number(food.calories) || 0;

        total.protein += Number(food.protein) || 0;

        total.carbs += Number(food.carbs) || 0;

        total.fat += Number(food.fat) || 0;

        return total;

      },
      {
        calories: 0,
        protein: 0,
        carbs: 0,
        fat: 0
      }
    );
  }


  /* =========================
     HELPERS
  ========================= */

  function formatNumber(number) {

    if (
      Number.isInteger(
        Number(number)
      )
    ) {
      return Number(number);
    }

    return Number(number)
      .toFixed(1);
  }


  function escapeHTML(text) {

    return String(text)
      .replaceAll("&", "&amp;")
      .replaceAll("<", "&lt;")
      .replaceAll(">", "&gt;")
      .replaceAll('"', "&quot;")
      .replaceAll("'", "&#039;");
  }


  function showToast(message) {

    const toast =
      document.getElementById("toast");

    toast.textContent = message;

    toast.classList.add("show");

    setTimeout(() => {

      toast.classList.remove("show");

    }, 1800);

  }


  /* =========================
     RENDER DASHBOARD
  ========================= */

  function render() {

    const totals = getTotals();

    const goals = data.goals;

    document.getElementById(
      "date"
    ).textContent =
      formatDate(currentDate);


    document.getElementById(
      "calories"
    ).textContent =
      Math.round(totals.calories);


    document.getElementById(
      "calorieGoalText"
    ).textContent =
      `of ${Math.round(goals.calories)} calories`;


    const remaining =
      Math.max(
        0,
        goals.calories -
        totals.calories
      );


    document.getElementById(
      "remaining"
    ).textContent =
      `${Math.round(remaining)} remaining`;


    const caloriePercent =
      goals.calories > 0
        ? Math.min(
            100,
            totals.calories /
            goals.calories *
            100
          )
        : 0;


    document.getElementById(
      "calorieProgress"
    ).style.width =
      caloriePercent + "%";


    document.getElementById(
      "protein"
    ).textContent =
      `${formatNumber(totals.protein)}g`;


    document.getElementById(
      "carbs"
    ).textContent =
      `${formatNumber(totals.carbs)}g`;


    document.getElementById(
      "fat"
    ).textContent =
      `${formatNumber(totals.fat)}g`;


    document.getElementById(
      "proteinGoal"
    ).textContent =
      formatNumber(goals.protein);


    document.getElementById(
      "carbsGoal"
    ).textContent =
      formatNumber(goals.carbs);


    document.getElementById(
      "fatGoal"
    ).textContent =
      formatNumber(goals.fat);


    renderFoodList();

  }


  /* =========================
     FOOD LIST
  ========================= */

  function renderFoodList() {

    const container =
      document.getElementById(
        "foodList"
      );

    const foods =
      getCurrentDay().foods;


    if (foods.length === 0) {

      container.innerHTML =
        `<div class="empty">
          Nothing logged yet.
        </div>`;

      return;
    }


    container.innerHTML =
      foods.map(food => {

        return `

          <div class="food">

            <div class="food-info">

              <div class="food-name">
                ${escapeHTML(food.name)}
              </div>

              <div class="food-macros">
                ${formatNumber(food.protein)}g P
                ·
                ${formatNumber(food.carbs)}g C
                ·
                ${formatNumber(food.fat)}g F
              </div>

            </div>

            <div class="food-calories">
              ${Math.round(food.calories)} kcal
            </div>

            <button
              class="food-button edit-button"
              data-id="${food.id}">
              Edit
            </button>

            <button
              class="food-button delete edit-button"
              data-delete-id="${food.id}">
              ✕
            </button>

          </div>

        `;

      }).join("");

  }

     FOOD MODAL

  function openFoodModal() {

  }


  function closeFoodModal() {

      "foodModal"
    ).classList.remove("active");

  }

      "editId"
    ).value = "";

      "foodCalories"

    document.getElementById(
      "foodCarbs"
    ).value = "";

    document.getElementById(
      "foodFat"
    ).value = "";


    document.getElementById(
      "foodModalTitle"
    ).textContent =
      "Add food";


    document.getElementById(
      "saveFoodButton"
    ).textContent =
      "Add food";

  }


  function openAddFood() {

    clearFoodForm();

    openFoodModal();

  }


  function openEditFood(id) {

    const food =
      getCurrentDay()
        .foods
        .find(
          food => food.id === id
        );


    if (!food) return;


    document.getElementById(
      "editId"
    ).value = id;


    document.getElementById(
      "foodName"
    ).value = food.name;


    document.getElementById(
      "foodCalories"
    ).value = food.calories;


    document.getElementById(
      "foodProtein"
    ).value = food.protein;


    document.getElementById(
      "foodCarbs"
    ).value = food.carbs;


    document.getElementById(
      "foodFat"
    ).value = food.fat;


    document.getElementById(
      "foodModalTitle"
    ).textContent =
      "Edit food";


    document.getElementById(
      "saveFoodButton"
    ).textContent =
      "Save changes";


    openFoodModal();

  }


  function getFoodFromForm() {

    const name =
      document.getElementById(
        "foodName"
      ).value.trim();


    if (!name) {

      showToast(
        "Enter a food name"
      );

      return null;

    }


    return {

      name,

      calories:
        Number(
          document.getElementById(
            "foodCalories"
          ).value
        ) || 0,

      protein:
        Number(
          document.getElementById(
            "foodProtein"
          ).value
        ) || 0,

      carbs:
        Number(
          document.getElementById(
            "foodCarbs"
          ).value
        ) || 0,

      fat:
        Number(
          document.getElementById(
            "foodFat"
          ).value
        ) || 0

    };

  }


  function saveFood(saveFavorite = false) {

    const food =
      getFoodFromForm();


    if (!food) return;


    const editId =
      document.getElementById(
        "editId"
      ).value;


    if (editId) {

      const index =
        getCurrentDay()
          .foods
          .findIndex(
            food =>
              food.id === editId
          );


      if (index !== -1) {

        getCurrentDay()
          .foods[index] = {
            ...getCurrentDay()
              .foods[index],
            ...food
          };

      }


      showToast(
        "Food updated"
      );

    } else {

      const newFood = {

        id:
          Date.now().toString(),

        ...food

      };


      getCurrentDay()
        .foods
        .push(newFood);


      if (saveFavorite) {

        data.savedFoods.push({
          id:
            Date.now().toString() +
            "-saved",

          ...food

        });

      }


      showToast(
        saveFavorite
          ? "Food added and saved"
          : "Food added"
      );

    }


    saveData();

    render();

    closeFoodModal();

  }


  /* =========================
     SAVED FOODS
  ========================= */

  function openSavedModal() {

    renderSavedFoods();

    document.getElementById(
      "savedModal"
    ).classList.add("active");

  }


  function renderSavedFoods() {

    const container =
      document.getElementById(
        "savedList"
      );


    if (
      data.savedFoods.length === 0
    ) {

      container.innerHTML =
        `<div class="empty">
          No saved foods yet.
        </div>`;

      return;

    }


    container.innerHTML =
      data.savedFoods
        .map(food => {

          return `

            <div class="saved-food">

              <div class="saved-info">

                <div class="saved-name">
                  ${escapeHTML(food.name)}
                </div>

                <div class="saved-details">
                  ${Math.round(food.calories)} kcal
                  · ${formatNumber(food.protein)}g P
                  · ${formatNumber(food.carbs)}g C
                  · ${formatNumber(food.fat)}g F
                </div>

              </div>

              <button
                class="use"
                data-use-id="${food.id}">
                Add
              </button>

              <button
                class="food-button delete"
                data-saved-delete="${food.id}">
                ✕
              </button>

            </div>

          `;

        })
        .join("");

  }


  function addSavedFood(id) {

    const food =
      data.savedFoods
        .find(
          food =>
            food.id === id
        );


    if (!food) return;


    getCurrentDay()
      .foods
      .push({

        id:
          Date.now().toString(),

        name: food.name,

        calories: food.calories,

        protein: food.protein,

        carbs: food.carbs,

        fat: food.fat

      });


    saveData();

    render();

    document.getElementById(
      "savedModal"
    ).classList.remove("active");


    showToast(
      "Saved food added"
    );

  }

  function openHistoryModal() {

    renderHistory();

    document.getElementById(
      "historyModal"
    ).classList.add("active");

  }


  function renderHistory() {

    const container =
      document.getElementById(
        "historyList"
      );


    const dates =
      Object.keys(
        data.days
      )
      .filter(
        date =>
          data.days[date]
            .foods
            .length > 0
      )
      .sort()
      .reverse();


    if (dates.length === 0) {

      container.innerHTML =
        `<div class="empty">
          No history yet.
        </div>`;

      return;

    }


    container.innerHTML =
      dates.map(date => {

        const foods =
          data.days[date].foods;


        const totals =
          foods.reduce(
            (total, food) => {

              total.calories +=
                Number(food.calories) || 0;

              total.protein +=
                Number(food.protein) || 0;

              return total;

            },
            {
              calories: 0,
              protein: 0
            }
          );


        return `

          <div class="history-day">

            <div class="history-info">

              <div class="history-name">
                ${formatDate(date)}
              </div>

              <div class="history-details">
                ${Math.round(totals.calories)} kcal
                · ${formatNumber(totals.protein)}g protein
                · ${foods.length} foods
              </div>

            </div>

            <button
              class="use"
              data-view-date="${date}">
              View
            </button>

          </div>

        `;

      }).join("");

  }

  /* =========================

  ========================= */

  document.addEventListener(
    "click",
    function(event) {

      const editButton =
        event.target.closest(
          "[data-id]"
        );


      if (
        editButton &&
        editButton.classList
          .contains("edit-button")
      ) {

        openEditFood(
          editButton.dataset.id
        );

        return;

      }


      const deleteButton =
        event.target.closest(
          "[data-delete-id]"
        );


      if (deleteButton) {

        const id =
          deleteButton.dataset
            .deleteId;


        getCurrentDay().foods =
          getCurrentDay()
            .foods
            .filter(
              food =>
                food.id !== id
            );


        saveData();

        render();

        showToast(
          "Food deleted"
        );

        return;

      }


      const useButton =
        event.target.closest(
          "[data-use-id]"
        );


      if (useButton) {

        addSavedFood(
          useButton.dataset.useId
        );

        return;

      }


      const savedDelete =
        event.target.closest(
          "[data-saved-delete]"
        );


      if (savedDelete) {

        const id =
          savedDelete.dataset
            .savedDelete;


        data.savedFoods =
          data.savedFoods
            .filter(
              food =>
                food.id !== id
            );


        saveData();

        renderSavedFoods();

        showToast(
          "Saved food deleted"
        );

        return;

      }


      const viewButton =
        event.target.closest(
          "[data-view-date]"
        );


      if (viewButton) {

        currentDate =
          viewButton.dataset
            .viewDate;


        document
          .getElementById(
            "historyModal"
          )
          .classList.remove(
            "active"
          );


        render();

      }

    }
  );
